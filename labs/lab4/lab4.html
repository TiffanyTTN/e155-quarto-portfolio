// ============================================================================
// 2. Local Functions
// ============================================================================

void updateAudioFrequency(int frequency) {
    // Avoid zero or inverted frequencies to prevent division faults
    if (frequency <= 0) {
        TIM2->CR1 &= ~TIM_CR1_CEN;     // Halt count engine
        digitalWrite(SPEAKER_PIN, GPIO_LOW); // Flatline output to fully eliminate speaker coil noise
        return;
    }
    
    TIM2->CR1 &= ~TIM_CR1_CEN;         // Freeze timer temporarily to change configuration parameters
    
    // Half-Period Auto-Reload Register Math: ARR = (Timer Clock Frequency / (2 * Frequency)) - 1
    // Timer frequency is exactly 1 MHz because the driver sets the prescaler to 15 (16MHz / 16)
    uint32_t arr_value = (1000000 / (2 * (uint32_t)frequency)) - 1;
    
    TIM2->ARR = arr_value;
    TIM2->CNT = 0;                     // Clear stale loop cycle values
    TIM2->EGR |= TIM_EGR_UG;           // Force immediate latch shadow reloads
    TIM2->SR  &= ~TIM_SR_UIF;          // Clear outstanding update flags
    TIM2->CR1 |= TIM_CR1_CEN;          // Re-engage countdown audio oscillator engine
}

void executionDelayMs(uint32_t ms) {
    uint32_t initial_timestamp = globalMillisecondsCounter;
    // Protect code against global counter variable overflow scenarios
    while ((globalMillisecondsCounter - initial_timestamp) < ms);
}

// ============================================================================
// 3. Exception Vector Targets
// ============================================================================

void SysTick_Handler(void) {
    globalMillisecondsCounter++; // Short, non-blocking asynchronous time heartbeat
}

void TIM2_IRQHandler(void) {
    if (TIM2->SR & TIM_SR_UIF) {
        TIM2->SR &= ~TIM_SR_UIF;       // Acknowledge milestone and reset update flag manually
        togglePin(SPEAKER_PIN);        // Execute portable inline toggle command
    }
}

// ============================================================================
// 4. Main Loop
// ============================================================================

int main(void) {
    // Generic hardware driver blocks from peripherals.h
    driverSystemClockInit16MHz();
    driverSysTickInit();
    
    // Initialize Timer 2 with a prescaler of 15 (creates a clean 1 MHz clock baseline from 16 MHz HSI)
    driverTimer2InterruptInit(15);
    
    // Configure target pins using GPIO
    pinMode(SPEAKER_PIN, GPIO_OUTPUT);
    
    // Resolve score pointer references based on selection constraints (used to choose which notes to play)
    const int (*active_score)[2] = (PLAYBACK_TRACK == 1) ? notes_ode_to_joy : notes_fur_elise; // PLAYBACK_TRACK Mux selector: 1 = Für Elise, 0 = Ode to Joy
    int current_note_idx = 0;
    
    while (1) {
        int pitch    = active_score[current_note_idx][0];
        int duration = active_score[current_note_idx][1];
        
        // Check for composition matrix structural termination flag {0, 0}
        if (duration == 0) {
            break; 
        }
        
        // Forward parameters down to safe configuration helpers
        updateAudioFrequency(pitch);
        executionDelayMs(duration);
        
        // Enforce brief silent guard window to make adjacent matching notes sound good
        TIM2->CR1 &= ~TIM_CR1_CEN;
        digitalWrite(SPEAKER_PIN, GPIO_LOW);
        executionDelayMs(NOTE_GAP_MS);
        
        current_note_idx++;
    }
    
    // Cleanly isolate speaker load and enter a safe lock state upon song completion
    TIM2->CR1 &= ~TIM_CR1_CEN;
    digitalWrite(SPEAKER_PIN, GPIO_LOW);
    while (1) {
        // Idle halt state
    }
}

// --- Core RCC Speed Drivers ---
static inline void driverSystemClockInit16MHz(void) {
    RCC->CR |= RCC_CR_HSION;
    while (!(RCC->CR & RCC_CR_HSIRDY)); 
    
    uint32_t cfg_temp = RCC->CFGR;
    cfg_temp &= ~(0x3 << 0);
    cfg_temp |= SW_HSI16;
    RCC->CFGR = cfg_temp;
    while (((RCC->CFGR >> 2) & 0x3) != SW_HSI16);
    
    uint32_t hpre_temp = RCC->CFGR;
    hpre_temp &= ~(0xF << 4);
    hpre_temp |= (HPRE_DIV1 << 4);
    RCC->CFGR = hpre_temp;
}

// --- General Purpose I/O Port B Drivers ---

// From GPIO.c tutorial code
static inline void pinMode(int pin, int function) {
    switch(function) {
        case GPIO_INPUT:
            GPIOB->MODER &= ~(0b11 << (2 * pin));
            break;
        case GPIO_OUTPUT:
            GPIOB->MODER |= (0b1 << (2 * pin));
            GPIOB->MODER &= ~(0b1 << (2 * pin + 1));
            break;
        case GPIO_ALT:
            GPIOB->MODER &= ~(0b1 << (2 * pin));
            GPIOB->MODER |= (0b1 << (2 * pin + 1));
            break;
        case GPIO_ANALOG:
            GPIOB->MODER |= (0b11 << (2 * pin));
            break;
    }
}

// From GPIO.c tutorial code
static inline void digitalWrite(int pin, int val) {
    if (val == GPIO_HIGH) {
        GPIOB->ODR |= (1 << pin);
    } else {
        GPIOB->ODR &= ~(1 << pin);
    }
}

// From GPIO.c tutorial code
static inline void togglePin(int pin) {
    GPIOB->ODR ^= (1 << pin);
}

// From GPIO.c tutorial code
static inline int digitalRead(int pin) {
    return (int)((GPIOB->IDR >> pin) & 1);
}

// --- Generic Time and Interrupt Peripherals Configurations ---
static inline void driverSysTickInit(void) {
    SysTick_Core->CTRL = 0;                                       
    SysTick_Core->LOAD = (SYSTEM_CLOCK_FREQ / 1000) - 1;             
    SysTick_Core->VAL  = 0;                                       
    SysTick_Core->CTRL = (SYSTICK_CLKSRC | SYSTICK_TICKINT | SYSTICK_ENABLE); 
}

static inline void driverTimer2InterruptInit(uint16_t prescalerValue) {
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOBEN;  
    RCC->APB1ENR1 |= RCC_APB1ENR1_TIM2EN; 
    
    TIM2->PSC = prescalerValue;                                               
    TIM2->DIER |= TIM_DIER_UIE;                                   
    
    // Set enable state safely inside the indexed array layout matrix
    NVIC_Core->ISER[NVIC_TIM2_IRQ_POS >> 5] |= (1 << (NVIC_TIM2_IRQ_POS & 0x1F));
}

Calculations

Documentation and calculations for pitches related to timer configuration: Pitch Generation Math (TIM2)

  1. To generate a pure square-wave tone at a specific target pitch frequency \(f_{pitch}\), the microcontroller pin must toggle twice per complete wave period (once from low-to-high, and once from high-to-low). Therefore, the timer update interrupt frequency \(f_{interrupt}\) must be double the target pitch frequency: \(f_{interrupt}\) = \(2 \times f_{pitch}\).
  2. From MCU reference manual for timers, \(f_{interrupt} = \frac{f_{clk}}{(PSC + 1)(ARR + 1)}\). Plugging in \(f_{interrupt}\) = \(2 \times f_{pitch}\), PSC = 15, \(f_{clk} = 16 MHz\), the equation can be rearranged to get \(ARR = (\frac{1,000,000}{2 \times f_{pitch}}) - 1\).

Documentation and calculations for duration related to timer configuration: Note Duration Math (SysTick)

  1. The system clock baseline is set to 16 MHz (16,000,000 ticks/second). To achieve a continuous 1 ms hardware time heartbeat, the reload threshold register (LOAD) is computed as: \(LOAD = (\frac{f_{clk}}{1000}) -1 = (\frac{16,000,000}{1000}) - 1 = 15,999\).
  2. This means that every 16,000,000 clock cycles, the SysTick timer rolls over and increments the global counter “globalMillisecondsCounter” inside the asynchronous SysTick_Handler of the code by exactly 1, which allows the linear duration math loop to match a clock time in milliseconds.

Accuracy Analysis for 220 Hz Pitch:

  1. For an ideal float evaluation, the ARR equation can be used. Plugging in \(f_{pitch}\) = 220 Hz, \(ARR_{ideal} = (\frac{1,000,000}{2 \times 220}) - 1 = (\frac{1,000,000}{440}) - 1 = 2271.727\).
  2. As an integer for hardware, \(ARR_{actual} = 2271\).
  3. An audio tone requires a full, completed wave cycle (going HIGH and then going back LOW). So, there are two interrupts: (1) toggling the pin from LOW to HIGH (creating the first half of the wave) and (2) toggling the pin from HIGH to LOW (creating the second half of the wave). Because it takes two distinct interrupts to make one full sound wave period, the frequency of the actual pitch \(f_{actual}\) is half the interrupt frequency: \(f_{actual} = \frac{f_{interrupt}}{2} = \frac{1,000,000}{2 \times (ARR +1)}\).
  4. Plugging in \(ARR_{actual}\) in \(f_{actual}\) equation, \(f_{actual} = \frac{1,000,000}{2 \times (2271 +1)} \approx 220.0704 Hz\).
  5. The accuracy error percentage calculation in this case is then \(Error = |\frac{220.0704 - 220}{220}| \times 100 \% \approx 0.032 \%\), which is below the max 1% error threshold limit.

Accuracy Analysis for 1000 Hz Pitch:

  1. For an ideal float evaluation, the ARR equation can be used. Plugging in \(f_{pitch}\) = 1000 Hz, \(ARR_{ideal} = (\frac{1,000,000}{2 \times 1000}) - 1 = (\frac{1,000,000}{2000}) - 1 = 499\).
  2. As an integer for hardware, \(ARR_{actual} = 499\).
  3. Recall that equation for the actual resulting output frequency is \(f_{actual} = \frac{f_{interrupt}}{2} = \frac{1,000,000}{2 \times (ARR +1)}\).
  4. Plugging in \(ARR_{actual}\) in \(f_{actual}\) equation, \(f_{actual} = \frac{1,000,000}{2 \times (499 +1)} = 1000 Hz\).
  5. The accuracy error percentage calculation in this case is then \(Error = |\frac{1000 - 1000}{1000}| \times 100 \% = 0.0 \%\), which is below the max 1% error threshold limit.

AI Prototype

LLM Prompt: What timers should I use on the STM32L432KC to generate frequencies ranging from 220Hz to 1kHz? What’s the best choice of timer if I want to easily connect it to a GPIO pin? What formulae are relevant, and what registers need to be set to configure them properly?

After prompting the LLM (Google Gemini) with the given prompt, it recommended to use the general-purpose timer TIM2 as the best option, or TIM15/TIM16 in PWM Mode 1 to to generate frequencies from 220 Hz to 1 kHz directly on a GPIO pin without CPU overhead. It also mentioned to avoid basic timers, such as TIM6 and TIM7, because they lack hardware output channels and cannot drive GPIO pins directly. To support its results, it referred to STMicroelectronics’s STM32L432KC resources and provided an example of easy GPIO connections for TIM2 and TIM15/TIM16. It also showed relevant formulas including the counter tick frequency, output PWM frequency, duty cycle via the Capture/Compare Register (CCRx). An example calculation then followed for when PSC = 3 such that the timer ticks cleanly at 1 MHz. More registers were provided and what the bits for each of them mean. After providing the same prompt but with the E155’s STM32L432KC reference manual, the same results were given.

Compared to my own read of the reference manual, the LLM was much faster in identifying and describing details, along with providing examples. This then makes it much quicker and easier to look for specific details than using Ctrl-F in the reference manual because the LLM summarizes information based on the provided prompt. I would then rate the quality of the output a 6/10 because it led to similar conclusions that I had when reviewing the reference manual. However, it only provided general resources on how it got its information and no specifications on which sections of the resources it was referring to to support its output. When reviewing the reference manual on my own, I try to document which sections I am referring to so its easier to look back at them later.
